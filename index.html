// ─── DHAN API CONFIG ─────────────────────────────────────────── const DHAN_TOKEN = localStorage.getItem("dhan_token") || ""; const DHAN_CLIENT = localStorage.getItem("dhan_client") || ""; const SECURITY_IDS = { TCS:"11536", INFY:"1594", RELIANCE:"2885", HDFCBANK:"1333", ICICIBANK:"4963", WIPRO:"3787", SBIN:"3045", SUNPHARMA:"3351", AXISBANK:"5900", HCLTECH:"7229", TATAMOTORS:"3456", MARUTI:"10999", LTIM:"17818", DRREDDY:"881", ONGC:"2475", NIFTY50:"13" }; // ─── FETCH LIVE QUOTE FROM DHAN ──────────────────────────────── async function fetchDhanQuote(symbol) { const secId = SECURITY_IDS[symbol]; if (!secId) throw new Error("Symbol not in Dhan ID list"); const response = await fetch(`https://api.dhan.co/v2/quotes/${secId}`, { method: "GET", headers: { "access-token": DHAN_TOKEN, "client-id": DHAN_CLIENT, "Content-Type": "application/json" } }); if (!response.ok) throw new Error("Dhan API error: " + response.status); return await response.json(); } // ─── FETCH HISTORICAL DATA FOR RS SCORE ──────────────────────── async function fetchDhanHistory(symbol, days = 365) { const secId = SECURITY_IDS[symbol]; if (!secId) throw new Error("Symbol not found"); const toDate = new Date().toISOString().split("T")[0]; const fromDate = new Date(Date.now() - days * 86400000) .toISOString().split("T")[0]; const response = await fetch("https://api.dhan.co/v2/charts/historical", { method: "POST", headers: { "access-token": DHAN_TOKEN, "client-id": DHAN_CLIENT, "Content-Type": "application/json" }, body: JSON.stringify({ securityId: secId, exchangeSegment: "NSE_EQ", instrument: "EQUITY", expiryCode: 0, oi: false, fromDate: fromDate, toDate: toDate }) }); if (!response.ok) throw new Error("History fetch failed"); return await response.json(); } // ─── FETCH INTRADAY FOR VWAP ──────────────────────────────────── async function fetchDhanIntraday(symbol) { const secId = SECURITY_IDS[symbol]; const today = new Date().toISOString().split("T")[0]; const response = await fetch("https://api.dhan.co/v2/charts/intraday", { method: "POST", headers: { "access-token": DHAN_TOKEN, "client-id": DHAN_CLIENT, "Content-Type": "application/json" }, body: JSON.stringify({ securityId: secId, exchangeSegment: "NSE_EQ", instrument: "EQUITY", interval: "5", fromDate: today, toDate: today }) }); return await response.json(); } // ─── CALCULATE VWAP FROM INTRADAY DATA ───────────────────────── function calculateVWAP(intradayData) { const candles = intradayData?.data || []; if (!candles.length) return null; let sumTPV = 0, sumVol = 0; candles.forEach(c => { const tp = (c.high + c.low + c.close) / 3; sumTPV += tp * c.volume; sumVol += c.volume; }); return sumVol > 0 ? sumTPV / sumVol : null; } // ─── MAIN FETCH FUNCTION USING DHAN ──────────────────────────── async function fetchLive(symbol) { if (!DHAN_TOKEN || !DHAN_CLIENT) { throw new Error("Dhan credentials missing. Add in Setup tab."); } // Fetch all 3 in parallel for speed const [quote, history, niftyHistory, intraday] = await Promise.all([ fetchDhanQuote(symbol), fetchDhanHistory(symbol, 380), fetchDhanHistory("NIFTY50", 380), fetchDhanIntraday(symbol) ]); // Parse close prices from history const closes = history?.close || history?.data?.map(d => d.close) || []; const volumes = history?.volume || history?.data?.map(d => d.volume) || []; const nc = niftyHistory?.close || niftyHistory?.data?.map(d => d.close) || []; const n = closes.length - 1; // ROC function const roc = (arr, f, t) => arr[t] && arr[f] ? (arr[t] - arr[f]) / arr[f] * 100 : 0; // RS Score vs NIFTY (O'Neil weighted formula) const A = roc(closes, Math.max(0, n-63), n); const B = roc(closes, Math.max(0, n-126), Math.max(0, n-63)); const C = roc(closes, Math.max(0, n-189), Math.max(0, n-126)); const D = roc(closes, Math.max(0, n-252), Math.max(0, n-189)); const nn = nc.length - 1; const NA = roc(nc, Math.max(0, nn-63), nn); const NB = roc(nc, Math.max(0, nn-126), Math.max(0, nn-63)); const NC = roc(nc, Math.max(0, nn-189), Math.max(0, nn-126)); const ND = roc(nc, Math.max(0, nn-252), Math.max(0, nn-189)); const RS_RAW = 0.4*(A-NA) + 0.2*(B-NB) + 0.2*(C-NC) + 0.2*(D-ND); const rsScore = Math.min(99, Math.max(1, Math.round((RS_RAW + 30) / 60 * 98 + 1) )); // EMA calculation const ema = (arr, p) => { const k = 2 / (p + 1); let e = arr[0]; for (let i = 1; i < arr.length; i++) e = arr[i] * k + e * (1 - k); return e; }; const cmp = quote?.data?.ltp || quote?.ltp || closes[n]; const e9 = ema(closes.slice(-20), 9); const e21 = ema(closes.slice(-30), 21); const e50 = ema(closes.slice(-60), 50); const e200 = ema(closes, 200); // RSI const sl = closes.slice(n - 14); const gains = [], losses = []; for (let i = 1; i < sl.length; i++) { const d = sl[i] - sl[i-1]; gains.push(d > 0 ? d : 0); losses.push(d < 0 ? -d : 0); } const ag = gains.reduce((a,b) => a+b, 0) / gains.length || 1; const al = losses.reduce((a,b) => a+b, 0) / losses.length || 0.001; const rsi = Math.round(100 - 100 / (1 + ag / al)); // Volume const vslice = volumes.slice(Math.max(0, n-20), n); const avgVol = vslice.reduce((a,b) => a+b, 0) / (vslice.length || 1) || 1; const vPct = Math.round((volumes[n] / avgVol - 1) * 100); // VWAP from real intraday data const vwap = calculateVWAP(intraday); const vwapScore = vwap ? (cmp > vwap ? (((cmp - vwap) / vwap * 100) >= 2 ? 90 : ((cmp - vwap) / vwap * 100) >= 1 ? 75 : 62) : 30) : (cmp > e21 ? 62 : 35); // Delivery volume from Dhan quote const delVol = quote?.data?.deliveryQty || 0; const totalVol = quote?.data?.totalTradedQty || volumes[n] || 1; const delRatio = Math.round((delVol / totalVol) * 100) || 52; const delScore = delRatio >= 60 ? 80 : delRatio >= 50 ? 65 : delRatio >= 40 ? 45 : 30; const emaRaw = (cmp > e9 ? 10 : 0) + (e9 > e21 ? 15 : 0) + (e21 > e50 ? 25 : 0) + (e50 > e200 ? 30 : 0); const fiiScore = Math.min(99, Math.round( (vPct > 20 ? 30 : vPct > 0 ? 18 : 5) + (cmp > e200 ? 38 : 10) + (rsi > 55 ? 18 : 5) )); const w52h = Math.round((cmp / (quote?.data?.fiftyTwoWeekHigh || cmp)) * 100); return { cmp: Math.round(cmp * 100) / 100, chg: Math.round((quote?.data?.percentChange || 0) * 100) / 100, rs: rsScore, del: delScore, ema: Math.min(99, Math.round(emaRaw / 80 * 99)), vwap: vwapScore, fii: fiiScore, rsi, sma50: cmp > e50 ? "above" : "below", sma200: cmp > e200 ? "above" : "below", macd: e9 > e21 ? "bullish" : e9 < e21 ? "bearish" : "neutral", adx: 22, vol: (vPct >= 0 ? "+" : "") + vPct + "%", w52h, delRatio, sector: "NSE", _live: true, _cl: closes.slice(-252), _nc: nc.slice(-252) }; }
